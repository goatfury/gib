import { readFile, writeFile, copyFile, mkdir } from 'node:fs/promises';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = dirname(fileURLToPath(import.meta.url));
const htmlPath = resolve(root, 'public/index.html');
const dataPath = resolve(root, 'refined/refined-data.json');
const cssSource = resolve(root, 'refined/refined-toggle.css');
const cssTarget = resolve(root, 'public/refined-toggle.css');
const publicDataTarget = resolve(root, 'public/data/refined-products.json');

await mkdir(resolve(root, 'public/data'), { recursive: true });
await copyFile(cssSource, cssTarget);
await copyFile(dataPath, publicDataTarget);

let html = await readFile(htmlPath, 'utf8');
if (html.includes('oil-refined-toggle-2026-09-01')) {
  console.log('Refined-products toggle already installed.');
  process.exit(0);
}
const refinedData = JSON.parse(await readFile(dataPath, 'utf8'));

function replaceOnce(needle, replacement, label) {
  const count = html.split(needle).length - 1;
  if (count !== 1) throw new Error(`${label}: expected one match, found ${count}`);
  html = html.replace(needle, replacement);
}
