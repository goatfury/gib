  function refinedReadingAt(view, date) {
    const items = refinedData[view].months;
    const monthStart = new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), 1));
    let item = items[0];
    for (const candidate of items) {
      if (dt(candidate.month) <= monthStart) item = candidate;
      else break;
    }
    const latest = items.at(-1);
    const held = monthStart > dt(latest.month);
    return { ...item, t:dt(item.month), held, latestMonth:latest.month };
  }

  function refinedRegionMixAt(date) {
    const reading = refinedReadingAt('world', date);
    return { ...reading.regions, total:reading.value, held:reading.held, latest:reading.latest };
  }
